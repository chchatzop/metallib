$noop(
MetalLib library layout -
  Artist [CC] / YYYY - Album [type] [edition] [RE yyyy] [catalog MEDIA] [quality] /
  Artist - Album [edition] [RE yyyy] - NN - Title
  where round brackets are written for type / edition / RE / CC.
Line breaks and tabs are ignored - spaces are not.
Needs the MetalLib Quality and MetalLib Naming plugins.
)

$noop(--- values ---)
$noop(Names: ASCII - colon becomes space-dash. An artist ending in a dot keeps it - A.N.I.M.A.L. -
in files and in the folder - the country code always follows it. Album and title keep an ending
... or initialism.)
$set(_artist_file,$clean($if2(%albumartist%,%artist%),all))
$set(_album,$clean(%album%,keep))
$set(_title,$clean($if2(%title%,%_filename%),title))
$noop(Reissue: a pressing at least this many years after the first release gets RE yyyy)
$set(_re_gap,2)
$noop(Country: the BAND's - from Metal Archives - else MusicBrainz's artist country - else XU for
unknown. International bands are XW. Never the release country - that is where a pressing was made.)
$set(_cc,$if2(%_ma_band_country_code%,$getmulti(%_albumartists_countries%,0),XU))
$noop(Folder year = first release - a later pressing year shows as RE or RM)
$set(_year,$left($if2(%originalyear%,%originaldate%,%date%),4))
$set(_pyear,$left(%date%,4))
$set(_later,$if($and(%_year%,%_pyear%,$gte($sub(%_pyear%,%_year%),%_re_gap%)),%_pyear%))
$set(_notes,$lower(%_releasecomment% %_releasepackaging% %media%))
$set(_types,$lower(%releasetype%))
$set(_rm,$if2(%remaster%,$if($in(%_notes%,remaster),%_later%)))
$set(_re,$if2(%reissue%,$if($not(%_rm%),%_later%)))

$noop(--- edition flags: your own edition tag or the release notes ---)
$set(_digipak,$if($or($in(%_notes%,digipak),$in(%_notes%,digipack)),1))
$set(_jap,$if($or($in(%edition%,Jap),$in(%_notes%,japan),$and($eq(%releasecountry%,JP),$ne(%_cc%,JP))),1))
$set(_lim,$if($or($in(%edition%,Lim),$in(%_notes%,limited)),1))
$set(_us,$if($in(%edition%,US),1))
$set(_dlx,$if($or($in(%edition%,Dlx),$in(%_notes%,deluxe)),1))
$set(_expdel,$if($in(%edition%,Exp Del),1))
$set(_exp,$if($or($in(%edition%,Expnd),$and($in(%_notes%,expanded),$not(%_expdel%))),1))

$noop(--- source bracket: WEB for digital - else catalog + MEDIA - else MEDIA ---)
$noop(A copy ripped from the web is WEB whatever its tags say - the folder name tells - else the album media)
$set(_rip,$folder_media(%_dirname%))
$set(_src,$if2($if($or($eq(%_rip%,WEB),$eq(%catnum%,WEB)),WEB),$media_code($album_media()),%_rip%,$media_code(%catnum%)))
$set(_bracket,$if($eq(%_src%,WEB),WEB,$with_media($clean($getmulti(%catalognumber%,0)),%_src%)))
$set(_q,$album_quality())

$noop(--- folders: artists straight in the destination folder - no letter folder - user ---)
%_artist_file% \(%_cc%\)/
$if(%_year%,%_year% - )%_album%
$if($in(%_types%,single), \(Single\))
$if($in(%_types%,ep), \(EP\))
$if($in(%_types%,demo), \(Demo\))
$if($in(%_types%,compilation), \(Compilation\))
$if($in(%_types%,live), \(Live\))
$if($in(%releasestatus%,bootleg), \(Bootleg\))
$if($in(%releasestatus%,promo), \(Promo\))
$if(%_digipak%, \(Digipak\))
$if(%_jap%, \(Jap. Ed.\))
$if(%_lim%, \(Lim. Ed.\))
$if(%_us%, \(US Ed.\))
$if(%_dlx%, \(Del. Ed.\))
$if(%_expdel%, \(Exp. Del. Ed.\))
$if(%_exp%, \(Exp. Ed.\))
$if(%_re%, \(RE %_re%\))
$if(%_rm%, \(RM %_rm%\))
$if(%_bracket%, [%_bracket%])
$if(%_q%, [%_q%])
/

$noop(--- file ---)
%_artist_file% - %_album%
$if($in(%_types%,single), \(Single\))
$if(%_dlx%, \(Del. Ed.\))
$if(%_jap%, \(Jap. Ed.\))
$if(%_lim%, \(Lim. Ed.\))
$if(%_us%, \(US. Ed.\))
$if(%_expdel%, \(Exp. Del. Ed.\))
$if(%_exp%, \(Exp. Ed.\))
$if(%_re%, \(RE %_re%\))
$if(%_rm%, \(RM %_rm%\))
 - $if($gt(%totaldiscs%,1),%discnumber%-)$num(%tracknumber%,2) - %_title%
